MODULE exit INPUT.

  CASE ok_code.

    WHEN 'EXIT'.
      LEAVE PROGRAM.

    WHEN 'CANC'.

      CASE sy-dynnr.
        WHEN '0100'.
          LEAVE TO SCREEN 0.

        WHEN '0200'.
          LEAVE TO SCREEN 100.

        WHEN '0300'.
          LEAVE TO SCREEN 100.
      ENDCASE.

  ENDCASE.

ENDMODULE.

MODULE user_command_0100 INPUT.
 CASE ok_code.
    WHEN 'BOOKN'.
      PERFORM create_booking.
      set SCREEN '0300'.
    WHEN 'BOOKC'.
      PERFORM cancel_booking.
      SET SCREEN '0200'.
    WHEN 'BACK'.
      SET SCREEN '0000'.
    WHEN OTHERS.
      SET SCREEN '0100'.
  ENDCASE.
ENDMODULE.
