FORM create_booking .
   DATA: lv_bookid TYPE zsbook_18-bookid,
        ls_book   TYPE zsbook_18.

  " # BOOKID ##
  SELECT MAX( bookid ) INTO lv_bookid FROM zsbook_18.
  IF sy-subrc <> 0 OR lv_bookid IS INITIAL.
    lv_bookid = 1.
  ELSE.
    lv_bookid = lv_bookid + 1.
  ENDIF.

  " Work area ###
  ls_book-carrid = gs_data-carrid.
  ls_book-connid = gs_data-connid.
  ls_book-fldate = gs_data-fldate.
  ls_book-bookid = lv_bookid.

  " Insert
  INSERT zsbook_18 FROM ls_book.

  IF sy-subrc = 0.
    COMMIT WORK.
    MESSAGE s003.
  ELSE.
    ROLLBACK WORK.
    MESSAGE e004.
  ENDIF.
ENDFORM.

FORM cancel_booking .

 DELETE FROM zsbook_18 WHERE carrid = gs_data-carrid AND
                              connid = gs_data-connid AND
                              fldate = gs_data-fldate.
  IF sy-subrc = 0.
    MESSAGE s005.
  ELSE.
    MESSAGE e006.
  ENDIF.

ENDFORM.

* Checks if the flight exists in SPFLI
FORM check_spfli.
  DATA: ls_sflight TYPE zsflight_18,
        ls_spfli   TYPE zspfli_18.
    SELECT SINGLE *
    FROM zspfli_18 INTO @ls_spfli
    WHERE carrid = @gs_data-carrid
    AND connid = @gs_data-connid.

  IF sy-subrc = 0.
    MOVE-CORRESPONDING ls_spfli to zsdyn_conn_18.

  ELSE.
    MESSAGE e002 WITH zsdyn_conn_18-carrid
                                 zsdyn_conn_18-connid
                                 zsdyn_conn_18-fldate.
  ENDIF.
ENDFORM.

* Checks if the flight date is valid in SFLIGHT
FORM check_sflight.
  DATA: ls_sflight TYPE zsflight_18,
        ls_spfli   TYPE zspfli_18.
 SELECT SINGLE *
    FROM zsflight_18 INTO @ls_sflight
    WHERE carrid = @gs_data-carrid
    AND connid = @gs_data-connid
    AND fldate = @gs_data-fldate.

  IF sy-subrc = 0.
    MOVE-CORRESPONDING ls_sflight to zsdyn_conn_18.

  ELSE.
    MESSAGE e002 WITH zsdyn_conn_18-carrid
                                 zsdyn_conn_18-connid
                                 zsdyn_conn_18-fldate.
  ENDIF.
ENDFORM.
